{{- /* Horizontal bar chart comparing one metric. Rendered at build time; no JavaScript. Params: title (required), unit (subtitle, e.g. "ms, lower is better"), suffix (appended to values, e.g. " ms"), better ("lower" default | "higher"), note. Body: one bar per line, "Name = value" with optional "| tooltip detail": {{< bar-chart title="Startup to first frame" unit="ms, lower is better" suffix=" ms" >}} Bubble Tea = 13.3 | CPU 12 ms Ratatui = 8.0 {{< /bar-chart >}} */ -}} {{- $title := .Get "title" -}} {{- $suffix := .Get "suffix" | default "" -}} {{- $higher := eq (.Get "better") "higher" -}} {{- /* Parse the body into rows. */ -}} {{- $rows := slice -}} {{- range $i, $line := split (strings.TrimSpace .Inner) "\n" -}} {{- $line = strings.TrimSpace $line -}} {{- if and $line (strings.Contains $line "=") -}} {{- $tip := "" -}} {{- with strings.Contains $line "|" -}} {{- $parts := split $line "|" -}} {{- $line = strings.TrimSpace (index $parts 0) -}} {{- $tip = strings.TrimSpace (delimit (after 1 $parts) "|") -}} {{- end -}} {{- $kv := split $line "=" -}} {{- $raw := strings.TrimSpace (index $kv 1) -}} {{- $rows = $rows | append (dict "name" (strings.TrimSpace (index $kv 0)) "raw" $raw "value" (float $raw) "tip" $tip) -}} {{- else if $line -}} {{- errorf "bar-chart %q: line %q must look like \"Name = value\" (%s)" $title $line $.Position -}} {{- end -}} {{- end -}} {{- if not $rows }}{{ errorf "bar-chart %q has no data (%s)" $title .Position }}{{ end -}} {{- $values := apply $rows "index" "." "value" -}} {{- $max := math.Max $values -}} {{- $best := cond $higher $max (math.Min $values) -}}
